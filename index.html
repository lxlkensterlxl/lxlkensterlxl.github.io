import os from langchain.chains import ConversationalRetrievalChain from langchain.memory import ConversationBufferMemory from langchain_community.document_loaders import TextLoader from langchain_community.vectorstores import Chroma from langchain_openai import OpenAIEmbeddings, ChatOpenAI from langchain.text_splitter import CharacterTextSplitter class AdaptiveAIEngine: def __init__(self, memory_file="user_profile.txt"): self.memory_file = memory_file self.embeddings = OpenAIEmbeddings() # Initialize or load local user profile/memory document if not os.path.exists(self.memory_file): with open(self.memory_file, "w") as f: f.write("Initial User Profile: The user is starting a new interaction history.\n") # Setup Vector Store for long-term semantic memory self.vectorstore = self._init_vectorstore() # Setup Conversational LLM and Short-term Memory self.llm = ChatOpenAI(model="gpt-4o", temperature=0.7) self.chat_memory = ConversationBufferMemory( memory_key="chat_history", return_messages=True ) self.qa_chain = ConversationalRetrievalChain.from_llm( llm=self.llm, retriever=self.vectorstore.as_retriever(search_kwargs={"k": 3}), memory=self.chat_memory ) def _init_vectorstore(self): loader = TextLoader(self.memory_file) documents = loader.load() text_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=50) docs = text_splitter.split_documents(documents) return Chroma.from_documents(docs, self.embeddings, persist_directory="./chroma_db") def learn_fact(self, new_fact: str): """Adds new long-term knowledge learned from the user.""" with open(self.memory_file, "a") as f: f.write(f"\nLearned Fact: {new_fact}") # Re-index vector store with new data loader = TextLoader(self.memory_file) documents = loader.load() text_splitter = CharacterTextSplitter(chunk_size=500, chunk_overlap=50) docs = text_splitter.split_documents(documents) self.vectorstore = Chroma.from_documents(docs, self.embeddings, persist_directory="./chroma_db") # Update chain retriever self.qa_chain.retriever = self.vectorstore.as_retriever(search_kwargs={"k": 3}) def chat(self, user_input: str) -> str: """Processes conversational input using retrieved long-term memory and short-term context.""" response = self.qa_chain.invoke({"question": user_input}) return response["answer"] # --- Example Usage --- if __name__ == "__main__": # Ensure OPENAI_API_KEY is set in your environment variables ai_engine = AdaptiveAIEngine() # Simulate conversation and learning print("AI: Hello! How can I assist you today?") # User teaches the AI something new user_input = "Remember that my favorite programming language is Python and I am building an iOS automation tool." ai_engine.learn_fact(user_input) print(f"User: {user_input}") print("AI: Got it! I've logged that your favorite language is Python and you're working on an iOS automation tool.") # Later query testing if it learned query = "What language should I use for my current project?" response = ai_engine.chat(query) print(f"User: {query}") print(f"AI: {response}")
